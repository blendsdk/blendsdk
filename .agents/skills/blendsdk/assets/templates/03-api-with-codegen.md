# Template: API with Codegen

> **Complexity**: Advanced
> **Packages Used**: webafx, postgresql, dbcore, expression, webafx-cache, codegen
> **Docker**: PostgreSQL, Redis
> **Features**: Database introspection, TypeScript type generation, Zod schema generation, CRUD from schema

## Overview

A full API that uses `codegen` to introspect your PostgreSQL database schema and automatically generate TypeScript interfaces, Zod validation schemas, and typed CRUD operations. Changes to the database schema are reflected in code by re-running the generator.

## Prerequisites

- Node.js >= 22.0.0
- Docker (for PostgreSQL and Redis)
- Yarn or npm

## File: src/scripts/generate.ts

```typescript
import { PostgreSQLDatabase } from 'blendsdk/postgresql';
import {
  SchemaContainer,
  PostgreSQLIntrospector,
  TypeGenerator,
  ZodGenerator,
} from 'blendsdk/codegen';
import { writeFileSync, mkdirSync } from 'fs';
import { resolve, dirname } from 'path';
import { fileURLToPath } from 'url';

const __dirname = dirname(fileURLToPath(import.meta.url));
const outDir = resolve(__dirname, '../generated');

async function generate() {
  const db = new PostgreSQLDatabase({
    host: 'localhost',
    port: 5432,
    database: '{{PROJECT_NAME}}',
    user: 'postgres',
    pass: 'postgres',
  });

  await db.connect();

  try {
    // Create a schema container and introspect the database
    const schema = new SchemaContainer();
    const introspector = new PostgreSQLIntrospector(db);
    await introspector.introspect(schema);

    // Generate TypeScript interfaces
    const tsGenerator = new TypeGenerator();
    const types = await tsGenerator.generate(schema);

    // Generate Zod schemas
    const zodGenerator = new ZodGenerator();
    const schemas = await zodGenerator.generate(schema);

    // Write output files
    mkdirSync(outDir, { recursive: true });
    writeFileSync(resolve(outDir, 'types.ts'), types);
    writeFileSync(resolve(outDir, 'schemas.ts'), schemas);

    console.log('✅ Generated types.ts and schemas.ts');
  } finally {
    await db.disconnect();
  }
}

generate().catch(console.error);
```

## File: src/controllers/product-controller.ts

```typescript
import { BaseController, NotFoundError } from 'blendsdk/webafx';
import { PostgreSQLDatabase } from 'blendsdk/postgresql';
import { query } from 'blendsdk/expression';
import { Request, Response } from 'express';
import { z } from 'zod';

// After running `yarn generate`, import from generated types:
// import type { Product } from '../generated/types.js';
// import { createProductSchema, updateProductSchema } from '../generated/schemas.js';

// Fallback schemas (replace with generated ones)
const createProductSchema = z.object({
  name: z.string().min(1).max(255),
  description: z.string().optional(),
  price: z.number().positive(),
  active: z.boolean().default(true),
});

const listSchema = z.object({
  page: z.coerce.number().int().min(1).default(1),
  limit: z.coerce.number().int().min(1).max(100).default(20),
});

export class ProductController extends BaseController {
  routes() {
    return [
      this.route().get('/').validate(listSchema).handle(this.list),
      this.route().get('/:id').handle(this.getById),
      this.route().post('/').secure().validate(createProductSchema).handle(this.create),
    ];
  }

  async list(req: Request, res: Response) {
    const { page, limit } = req.services.getParams<{ page: number; limit: number }>();
    const db = await req.services.get<PostgreSQLDatabase>('db');
    const offset = (page - 1) * limit;

    const data = await db.executeQuery<{ id: number; name: string; price: number }>(
      'SELECT * FROM products ORDER BY id LIMIT :limit OFFSET :offset',
      { limit, offset }
    );
    const countResult = await db.executeQuery<{ total: string }>(
      'SELECT COUNT(*) as total FROM products'
    );
    const total = Number(countResult.records[0]?.total ?? 0);
    this.paginated(res, data.records, total, page, limit);
  }

  async getById(req: Request, res: Response) {
    const { id } = req.services.getParams<{ id: string }>();
    const db = await req.services.get<PostgreSQLDatabase>('db');
    const filter = query().where('id').equals(id).compile();
    const result = await db.from('products').select().byExpression(filter).execute();

    if (!result || result.records.length === 0) {
      throw new NotFoundError(`Product ${id} not found`);
    }
    this.ok(res, result.records[0]);
  }

  async create(req: Request, res: Response) {
    const params = req.services.getParams<z.infer<typeof createProductSchema>>();
    const db = await req.services.get<PostgreSQLDatabase>('db');
    const result = await db.insert('products').values(params).returning('*').execute();
    this.created(res, result!.records[0]);
  }
}
```

## File: src/index.ts

```typescript
import { WebApplication } from 'blendsdk/webafx';
import { PostgreSQLDatabase } from 'blendsdk/postgresql';
import { memoryCachePlugin } from 'blendsdk/webafx-cache';
import { ProductController } from './controllers/product-controller.js';

const app = new WebApplication({
  PORT: 3000,
  ENV_MODE: 'development',
  LOG_LEVEL: 'INFO',
  CORS: true,
});

app.registerService({
  name: 'db',
  type: 'singleton',
  factory: async (container, settings) => {
    const db = new PostgreSQLDatabase({
      host: settings.get('DB_HOST', 'localhost'),
      port: settings.get('DB_PORT', 5432),
      database: settings.get('DB_NAME', '{{PROJECT_NAME}}'),
      user: settings.get('DB_USER', 'postgres'),
      pass: settings.get('DB_PASSWORD', 'postgres'),
    });
    await db.connect();
    return db;
  },
  dispose: async (db) => await db.disconnect(),
});

app.use(memoryCachePlugin({ rootKey: '{{PROJECT_NAME}}', defaultTTL: 300 }));
app.registerController('/api/products', ProductController);

const shutdown = await app.start();
```

## Codegen Workflow

```bash
# 1. Modify your database schema (add tables, columns, etc.)
# 2. Re-run the generator
yarn generate

# 3. Update controllers to use new generated types
# 4. TypeScript compiler catches any mismatches
```

## Customization Guide

1. **Custom type mappings** — Pass a custom `ColumnMapper` function to `introspector.introspect(schema, mapper)` to override PostgreSQL-to-TypeScript type conversions
2. **Table filtering** — Filter introspection results by schema name using the `SchemaContainer.scope()` method
3. **OpenAPI generation** — Use `OpenAPIGenerator` to produce API documentation from your controller routes
4. **CI integration** — Run `yarn generate` in CI to verify types are up-to-date
